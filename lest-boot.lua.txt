local function makeButton()
    if G.__LestBtn then pcall(function() G.__LestBtn:Destroy() end) end
    local UIS = game:GetService("UserInputService")
    local parent
    pcall(function() if gethui then parent = gethui() end end)
    if not parent then pcall(function() parent = game:GetService("CoreGui") end) end
    local gui = Instance.new("ScreenGui")
    gui.Name = "LestPanelBtn"
    gui.ResetOnSpawn = false
    gui.DisplayOrder = 999
    gui.IgnoreGuiInset = true
    pcall(function() if syn and syn.protect_gui then syn.protect_gui(gui) end end)
    local okParent = parent and pcall(function() gui.Parent = parent end)
    if not okParent or not gui.Parent then
        gui.Parent = Players.LocalPlayer:FindFirstChildOfClass("PlayerGui")
    end
    G.__LestBtn = gui

    local btn = Instance.new("TextButton")
    btn.Size = UDim2.fromOffset(46, 46)
    btn.Position = UDim2.new(0, 14, 0.5, -23)
    btn.BackgroundColor3 = Color3.fromHex("201017")
    btn.AutoButtonColor = false
    btn.Text = "L"
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 22
    btn.TextColor3 = Color3.fromHex("FFF0F2")
    btn.Parent = gui
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 14)
    corner.Parent = btn
    local stroke = Instance.new("UIStroke")
    stroke.Color = Color3.fromHex("FA526A")
    stroke.Thickness = 1.5
    stroke.Parent = btn
    local grad = Instance.new("UIGradient")
    grad.Color = ColorSequence.new(Color3.fromHex("2D1720"), Color3.fromHex("140B10"))
    grad.Rotation = 90
    grad.Parent = btn

    local mine = { open = true }
    local function toggle()
        local s = G.__LestOpus
        local win = s and s.Window
        if not win then return notify("Menu belum siap, tunggu sebentar") end
        local closed = win.Closed
        if type(closed) ~= "boolean" then closed = not mine.open end
        local ok = pcall(function()
            if closed then win:Open() else win:Close() end
        end)
        if ok then mine.open = closed else pcall(function() win:Toggle() end) end
    end
    G.__LestToggle = toggle

    local dragging, moved, dragStart, startPos = false, false, nil, nil
    btn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging, moved = true, false
            dragStart, startPos = input.Position, btn.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if not moved then toggle() end
                end
            end)
        end
    end)
    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
            local d = input.Position - dragStart
            if d.Magnitude > 6 then moved = true end
            if moved then
                btn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
            end
        end
    end)
    return mine
end

local okBtn, mine = pcall(makeButton)
if not okBtn then notify("Tombol panel gagal dibuat: " .. tostring(mine)) mine = { open = true } end

local function matchCase(m)
    if m:upper() == m then return "LEST" end
    if m:lower() == m then return "lest" end
    return "Lest"
end
local function rename(s)
    s = s:gsub("gentasaja", "\1KEEP\1")
    s = s:gsub("[Gg][Ee][Nn][Tt][Aa]%s?[Hh]4[Xx]", matchCase)
    s = s:gsub("[Gg][Ee][Nn][Tt][Aa]", matchCase)
    s = s:gsub("\1KEEP\1", "gentasaja")
    return s
end

local function replaceOnce(s, old, new)
    local i, j = s:find(old, 1, true)
    if not i then return nil end
    return s:sub(1, i - 1) .. new .. s:sub(j + 1)
end

local function attempt()

    local original = src
    local failed = false
    local function patch(old, new, required)
        local r = replaceOnce(src, old, new)
        if r then src = r elseif required then failed = true end
    end

    local a1 = "local function readUpvalues(fn)"
    patch(a1, [==[
local function pickLargestDatabase(match)
        local best, bestCount, report = nil, -1, {}
        for index, candidate in ipairs(match.Candidates) do
            local count = 0
            for id in pairs(candidate.Value) do
                if type(id) == "number" and id >= 0 and id % 1 == 0 then count += 1 end
            end
            report[#report + 1] = ("#%d=%d"):format(index, count)
            if count > bestCount then best, bestCount = candidate, count end
        end
        if best then
            Session.ModuleScan.DatabaseResolution = "entri terbanyak (" .. table.concat(report, ", ") .. ")"
            warn("[Lest] ItemDB ambigu, memakai kandidat terbesar: " .. table.concat(report, ", "))
        end
        return best
    end
    ]==] .. a1, true)

    patch("local function inspectValue(target, value, object)",
        "local function inspectValueSlow(target, value, object)", true)
    local a3 = "local filteredMatches, filteredTried = {}, false"
    patch(a3, [==[
local function inspectValue(target, value, object)
        if type(value) ~= "table" then return end
        local sample
        for i = 0, 6 do
            local s = rawget(value, i)
            if s ~= nil then sample = s break end
        end
        local maybeItems = type(sample) == "table" and type(rawget(sample, "name")) == "string"
        local maybeApi = rawget(value, "call") or rawget(value, "new") or rawget(value, "mount")
            or rawget(value, "isDeveloper") or rawget(value, "play")
        if not maybeItems and not maybeApi then return end
        return inspectValueSlow(target, value, object)
    end
    ]==] .. a3, true)

local best0 = pickLargestDatabase(filtered)
                if best0 then return best0.Value end

local value
