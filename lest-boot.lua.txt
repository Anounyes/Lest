G.__LestUI = { open = false, wantOpen = false, touched = false }

local function makeButton()
    if G.__LestBtn then pcall(function() G.__LestBtn:Destroy() end) end
    if G.__LestInputConn then pcall(function() G.__LestInputConn:Disconnect() end) end

    local UIS = game:GetService("UserInputService")
    local HttpService = game:GetService("HttpService")
    local SIZE = 48
    local POS_FILE = "LestOpus/button.json"

    local function viewport()
        local cam = workspace.CurrentCamera
        local v = cam and cam.ViewportSize
        if v and v.X > 0 then return v.X, v.Y end
        return 800, 400
    end
    local function loadPos()
        if type(isfile) ~= "function" or type(readfile) ~= "function" then return nil end
        local ok, raw = pcall(function() if isfile(POS_FILE) then return readfile(POS_FILE) end end)
        if not ok or not raw then return nil end
        local ok2, data = pcall(HttpService.JSONDecode, HttpService, raw)
        if ok2 and type(data) == "table" and type(data.x) == "number" and type(data.y) == "number" then return data end
    end
    local function savePos(xs, ys)
        if type(writefile) ~= "function" then return end
        pcall(function()
            if type(isfolder) == "function" and type(makefolder) == "function" and not isfolder("LestOpus") then
                makefolder("LestOpus")
            end
            writefile(POS_FILE, HttpService:JSONEncode({ x = xs, y = ys }))
        end)
    end

    local parent
    pcall(function() if gethui then parent = gethui() end end)
    if not parent then pcall(function() parent = game:GetService("CoreGui") end) end
    local gui = Instance.new("ScreenGui")
    gui.Name = "LestPanelBtn"
    gui.ResetOnSpawn = false
    gui.DisplayOrder = 999
    gui.IgnoreGuiInset = true
    pcall(function() if syn and syn.protect_gui then syn.protect_gui(gui) end end)
    local okParent = parent and pcall(function() gui.Parent = parent end)
    if not okParent or not gui.Parent then
        local pg = Players.LocalPlayer:FindFirstChildOfClass("PlayerGui") or Players.LocalPlayer:WaitForChild("PlayerGui", 10)
        gui.Parent = pg
    end
    G.__LestBtn = gui

    local vw, vh = viewport()
    local saved = loadPos()
    local px = saved and math.clamp(saved.x * vw, 0, vw - SIZE) or 14
    local py = saved and math.clamp(saved.y * vh, 0, vh - SIZE) or (vh / 2 - SIZE / 2)

    local btn = Instance.new("TextButton")
    btn.Size = UDim2.fromOffset(SIZE, SIZE)
    btn.Position = UDim2.fromOffset(px, py)
    btn.BackgroundColor3 = Color3.new(1, 1, 1)
    btn.AutoButtonColor = false
    btn.Text = "L"
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 24
    btn.TextColor3 = Color3.new(1, 1, 1)
    btn.TextStrokeColor3 = Color3.fromHex("5B1A6B")
    btn.TextStrokeTransparency = 0.5
    btn.Parent = gui

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 16)
    corner.Parent = btn
    local grad = Instance.new("UIGradient")
    grad.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Color3.fromHex("FF4D8D")),
        ColorSequenceKeypoint.new(0.5, Color3.fromHex("FF7A45")),
        ColorSequenceKeypoint.new(1, Color3.fromHex("B14DFF")),
    })
    grad.Rotation = 45
    grad.Parent = btn
    local stroke = Instance.new("UIStroke")
    stroke.Color = Color3.new(1, 1, 1)
    stroke.Transparency = 0.35
    stroke.Thickness = 2
    stroke.Parent = btn

    local function toggle()
        local ui = G.__LestUI
        ui.touched = true
        local s = G.__LestOpus
        local win = s and s.Window
        if not win then
            ui.wantOpen = true
            return notify("Menu sedang dimuat, akan terbuka sendiri saat siap")
        end
        local closed = win.Closed
        if type(closed) ~= "boolean" then closed = not ui.open end
        local ok = pcall(function()
            if closed then win:Open() else win:Close() end
        end)
        if ok then ui.open = closed end
    end

    local dragInput, dragStart, startX, startY, moved
    btn.InputBegan:Connect(function(input)
        if dragInput then return end
        local t = input.UserInputType
        if t == Enum.UserInputType.Touch or t == Enum.UserInputType.MouseButton1 then
            dragInput, moved = input, false
            dragStart = input.Position
            startX, startY = btn.AbsolutePosition.X, btn.AbsolutePosition.Y
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End and dragInput == input then
                    dragInput = nil
                    if moved then
                        local w, h = viewport()
                        savePos(btn.AbsolutePosition.X / w, btn.AbsolutePosition.Y / h)
                    else
                        toggle()
                    end
                end
            end)
        end
    end)
    G.__LestInputConn = UIS.InputChanged:Connect(function(input)
        if not dragInput then return end
        local t = input.UserInputType
        if input == dragInput or (t == Enum.UserInputType.MouseMovement and dragInput.UserInputType == Enum.UserInputType.MouseButton1) then
            local d = input.Position - dragStart
            if d.Magnitude > 6 then moved = true end
            if moved then
                local w, h = viewport()
                btn.Position = UDim2.fromOffset(
                    math.clamp(startX + d.X, 0, w - SIZE),
                    math.clamp(startY + d.Y, 0, h - SIZE))
            end
        end
    end)
end

local okBtn, errBtn = pcall(makeButton)
if not okBtn then notify("Tombol panel gagal dibuat: " .. tostring(errBtn)) end

local function matchCase(m)
    if m:upper() == m then return "LEST" end
    if m:lower() == m then return "lest" end
    return "Lest"
end
local function rename(s)
    s = s:gsub("gentasaja", "\1KEEP\1")
    s = s:gsub("[Gg][Ee][Nn][Tt][Aa]%s?[Hh]4[Xx]", matchCase)
    s = s:gsub("[Gg][Ee][Nn][Tt][Aa]", matchCase)
    s = s:gsub("\1KEEP\1", "gentasaja")
    return s
end

local function replaceOnce(s, old, new)
    local i, j = s:find(old, 1, true)
    if not i then return nil end
    return s:sub(1, i - 1) .. new .. s:sub(j + 1)
end

    local failed = false

        local r = replaceOnce(src, old, new)

    -- Pemilih ItemDB terbesar
    local a1 = "local function readUpvalues(fn)"
    patch(a1, [==[
local function pickLargestDatabase(match)
        local best, bestCount, report = nil, -1, {}
        for index, candidate in ipairs(match.Candidates) do
            local count = 0
            for id in pairs(candidate.Value) do
                if type(id) == "number" and id >= 0 and id % 1 == 0 then count += 1 end
            end
            report[#report + 1] = ("#%d=%d"):format(index, count)
            if count > bestCount then best, bestCount = candidate, count end
        end
        if best then
            Session.ModuleScan.DatabaseResolution = "entri terbanyak (" .. table.concat(report, ", ") .. ")"
            warn("[Lest] ItemDB ambigu, memakai kandidat terbesar: " .. table.concat(report, ", "))
        end
        return best
    end
    ]==] .. a1, true)

        if type(value) ~= "table" then return end
        local sample
        for i = 0, 6 do
            local s = rawget(value, i)
            if s ~= nil then sample = s break end
        end

            or rawget(value, "isDeveloper") or rawget(value, "play")

local value

            end
            value = databaseFromActions()]==], false)

    patch("local function structuralModule(name, retry)",
        "local function structuralModuleRaw(name, retry)", true)

local function structuralModule(name, retry)
        local key = "__LestCache_" .. name
        local cached = G[key]
        if cached then
            local okc, yes = pcall(matches, name, cached)
            if okc and yes then return cached end
        end

        if value then G[key] = value end
        return value, object, reason

    ]==] .. a5, true)

print("[GentaH4x] Memuat build " .. Session.Build)]==], false)

WindUI = downloadWindUI()()
