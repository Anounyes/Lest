local function matchCase(m)
    if m:upper() == m then return "LEST" end
    if m:lower() == m then return "lest" end
    return "Lest"
end
local function rename(s)
    s = s:gsub("gentasaja", "\1KEEP\1")
    s = s:gsub("[Gg][Ee][Nn][Tt][Aa]%s?[Hh]4[Xx]", matchCase)
    s = s:gsub("[Gg][Ee][Nn][Tt][Aa]", matchCase)
    s = s:gsub("\1KEEP\1", "gentasaja")
    return s
end

local function replaceOnce(s, old, new)
    local i, j = s:find(old, 1, true)
    if not i then return nil end
    return s:sub(1, i - 1) .. new .. s:sub(j + 1)
end
