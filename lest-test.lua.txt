local function alert(msg)

    pcall(function()

local function hex(h)
    local ok, c = pcall(Color3.fromHex, h)
    if ok and typeof(c) == "Color3" then return c end
    return Color3.fromRGB(tonumber(h:sub(1, 2), 16) or 255, tonumber(h:sub(3, 4), 16) or 255, tonumber(h:sub(5, 6), 16) or 255)
end

local function toggleMenu()
    local ui = G.__LestUI
    ui.touched = true
    local win = G.__LestOpus and G.__LestOpus.Window
    if not win then
        if not ui.wantOpen then alert("Menu masih dimuat, tunggu sebentar") end
        ui.wantOpen = true
        return
    end
    local closed = win.Closed
    if type(closed) ~= "boolean" then closed = not ui.open end
    pcall(function() if closed then win:Open() else win:Close() end end)
    ui.open = closed
end

local function makeButton()
    G.__LestToggle = nil
    if G.__LestInputConn then pcall(function() G.__LestInputConn:Disconnect() end) end
    if G.__LestBtn then pcall(function() G.__LestBtn:Destroy() end) end
    local UIS = game:GetService("UserInputService")
    local HttpService = game:GetService("HttpService")
    local TweenService = game:GetService("TweenService")
    local SIZE, POS_FILE = 54, "LestOpus/button.json"

    local function viewport()
        local cam = workspace.CurrentCamera
        local v = cam and cam.ViewportSize
        if v and v.X > 0 then return v.X, v.Y end
        return 800, 400
    end
    local function loadPos()
        local ok, data = pcall(function()
            if isfile and readfile and isfile(POS_FILE) then
                return HttpService:JSONDecode(readfile(POS_FILE))
            end
        end)
        if ok and type(data) == "table" and type(data.x) == "number" then return data end
    end
    local function savePos(x, y)
        pcall(function()
            if makefolder and isfolder and not isfolder("LestOpus") then makefolder("LestOpus") end
            writefile(POS_FILE, HttpService:JSONEncode({ x = x, y = y }))
        end)
    end

    local parent
    pcall(function() parent = gethui() end)
    if not parent then pcall(function() parent = game:GetService("CoreGui") end) end
    pcall(function()
        local old = parent and parent:FindFirstChild("LestPanelBtn")
        if old then old:Destroy() end
    end)
    local gui = Instance.new("ScreenGui")
    gui.Name, gui.ResetOnSpawn, gui.DisplayOrder, gui.IgnoreGuiInset = "LestPanelBtn", false, 999, true
    pcall(function() if syn and syn.protect_gui then syn.protect_gui(gui) end end)
    local placed = parent and pcall(function() gui.Parent = parent end)
    if not placed or not gui.Parent then
        local lp = game:GetService("Players").LocalPlayer
        gui.Parent = lp:FindFirstChildOfClass("PlayerGui") or lp:WaitForChild("PlayerGui", 10)
    end
    G.__LestBtn = gui

    local vw, vh = viewport()
    local saved = loadPos()
    local px = saved and math.clamp(saved.x * vw, 0, vw - SIZE) or 14
    local py = saved and math.clamp(saved.y * vh, 0, vh - SIZE) or (vh / 2 - SIZE / 2)

    -- Lencana bulat: dasar + cincin gradasi berputar
    local btn = Instance.new("TextButton")
    btn.Size, btn.Position = UDim2.fromOffset(SIZE, SIZE), UDim2.fromOffset(px, py)
    btn.BackgroundColor3, btn.AutoButtonColor, btn.Text = hex(P.tile), false, ""
    btn.Parent = gui

    local scale = Instance.new("UIScale")
    scale.Parent = btn

    pcall(function()
        local round = Instance.new("UICorner")
        round.CornerRadius = UDim.new(1, 0)
        round.Parent = btn

        local stroke = Instance.new("UIStroke")
        stroke.Color, stroke.Thickness = Color3.new(1, 1, 1), 2.5
        stroke.Parent = btn
        local rim = Instance.new("UIGradient")
        rim.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, hex(P.rimA)),
            ColorSequenceKeypoint.new(0.5, hex(P.rimB)),

        inner.Parent = btn
        local innerRound = Instance.new("UICorner")
        innerRound.CornerRadius = UDim.new(1, 0)
        innerRound.Parent = inner
        local innerStroke = Instance.new("UIStroke")
        innerStroke.Color, innerStroke.Thickness = hex(P.innerRim), 1

local FAST = [==[
local function inspectValue(target, value, object)
        if type(value) ~= "table" then return end
        local items, looked = false, 0
        for _, entry in pairs(value) do
            looked += 1
            if type(entry) == "table" and type(rawget(entry, "name")) == "string" then items = true break end
            if looked >= 4 then break end
        end
        local api = rawget(value, "call") or rawget(value, "new") or rawget(value, "mount")
            or rawget(value, "isDeveloper") or rawget(value, "play") or rawget(value, "add") or rawget(value, "on")
        if not items and not api then return end
        return inspectValueSlow(target, value, object)
    end
]==]

        local key = "__LestCache_" .. tostring(game.JobId) .. "_" .. name
        local cached = G[key]
        if cached then
            local okc, yes = pcall(matches, name, cached)
            if okc and yes then return cached end
        end
        local value, object, reason = structuralModuleRaw(name, retry)
        if value == nil and name == "ItemDB" then
            local best, bestN = nil, -1
            local sources = { runtimeMatches and runtimeMatches[name] or false, filteredMatches and filteredMatches[name] or false }
            for _, m in ipairs(sources) do
                for _, cand in ipairs(type(m) == "table" and m.Candidates or {}) do
                    local n = 0
                    for id in pairs(cand.Value) do
                        if type(id) == "number" then n += 1 end
                    end
                    if n > bestN then best, bestN = cand, n end
                end
            end
            if best then value, object, reason = best.Value, best.Object, nil end
        end
        if value ~= nil then G[key] = value end
        return value, object, reason
    end

                    end
                end
            end)
            if okc and cached then return cached end
            local data = rawDownload(url)
            pcall(function()
                if type(data) == "string" and #data > 1000 then
                    if type(makefolder) == "function" and type(isfolder) == "function" and not isfolder("LestOpus") then makefolder("LestOpus") end
                    writefile(dataFile, data)
                    writefile(timeFile, tostring(os.time()))
                end
            end)
            return data
        end
    end
end

local function rename(s)
    s = s:gsub("gentasaja", "\1KEEP\1")
    s = s:gsub("[Gg][Ee][Nn][Tt][Aa]%s?[Hh]4[Xx]", "Lest")
    s = s:gsub("[Gg][Ee][Nn][Tt][Aa]", "Lest")
    s = s:gsub("\1KEEP\1", "gentasaja")
    return s
end

local function buildSource(src)
    local out, applied, skipped = src, {}, {}
    local function swap(label, old, new)
        local i, j = out:find(old, 1, true)
        if i then
            out = out:sub(1, i - 1) .. new .. out:sub(j + 1)
            applied[#applied + 1] = label
        else
            skipped[#skipped + 1] = label
        end
    end
    local function at(text)
        return out:find(text, 1, true) or math.huge
    end

    local a = at("local function inspectValue(target, value, object)")
    local b = at("local filteredMatches, filteredTried = {}, false")
    if a < b and b < math.huge then
        swap("inspect", "local function inspectValue(target, value, object)", "local function inspectValueSlow(target, value, object)")
        swap("inspect-cepat", "local filteredMatches, filteredTried = {}, false", FAST .. "local filteredMatches, filteredTried = {}, false")
    end

    local c = at("local function structuralModule(name, retry)")
    local d = at("local function loadModule(parent, name, required)")
    if c < d and d < math.huge then
        swap("struktur", "local function structuralModule(name, retry)", "local function structuralModuleRaw(name, retry)")
        swap("cadangan-ItemDB", "local function loadModule(parent, name, required)", WRAP .. "local function loadModule(parent, name, required)")
    end

    swap("jeda", "if visited % 1000 == 0 then task.wait() end", "if visited % 250 == 0 then task.wait() end")
    if LITE then
        local lite = "if (Config.Data.Version == nil or Config.Data.Version == 1) and State.AfterBreakMs == 300 then"
        swap("ringan", lite, "pcall(function() State.LiteMode = true; State.Visual = false; State.MutePunch = true end)\n" .. lite)
    end
    swap("status-judul", "WindUI = downloadWindUI()()", UIBLOCK)
    local printLine = 'print("[GentaH4x] Memuat build " .. Session.Build)'
    swap("cache-windui", printLine, CACHE .. printLine)
    return out, applied, skipped
end

local function runMain()
    local okGet, src = pcall(function()
        return game:HttpGet(BASE .. MAIN_FILE .. "?t=" .. tostring(os.time()))
    end)
    if not okGet or type(src) ~= "string" or #src < 1000 then
        error("gagal unduh script utama: " .. tostring(src):sub(1, 120), 0)
    end
    local patched, applied, skipped = buildSource(src)
    warn("[Lest] tambalan: " .. table.concat(applied, ", ") .. " | dilewati: " .. table.concat(skipped, ", "))
    local fn, err = loadstring(rename(patched), "lest-main")
    if not fn then
        warn("[Lest] tambalan gagal dikompilasi, memakai versi asli: " .. tostring(err))
        fn, err = loadstring(rename(src), "lest-main")
        if not fn then error("script utama gagal dikompilasi: " .. tostring(err), 0) end
    end
    local ran, res = xpcall(fn, debug.traceback)
    if not ran then error(res, 0) end
end
